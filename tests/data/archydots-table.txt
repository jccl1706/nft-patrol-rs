table inet filter { # handle 1
	chain input { # handle 1
		type filter hook input priority filter; policy drop;
		ct state invalid drop comment "early drop of invalid connections" # handle 3
		ct state { established, related } accept comment "allow tracked connections" # handle 5
		iif "lo" accept comment "allow from loopback" # handle 6
		meta l4proto { icmp, ipv6-icmp } accept comment "allow icmp (IPv6 needs it to work at all)" # handle 8
		ip6 saddr fe80::/10 udp dport 546 accept comment "DHCPv6 replies" # handle 9
		ip saddr 192.168.10.0/24 tcp dport 22 accept comment "sshd - home LAN only" # handle 10
		meta pkttype host limit rate 5/second burst 5 packets counter reject with icmpx admin-prohibited # handle 11
		counter # handle 12
	}

	chain forward { # handle 2
		type filter hook forward priority filter; policy drop;
	}
}
